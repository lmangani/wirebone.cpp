function(wb_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE wirebone)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../src ${CMAKE_CURRENT_SOURCE_DIR}/../include)
  add_test(NAME ${name} COMMAND ${name})
endfunction()

wb_test(test_crypto)
wb_test(test_keys)
wb_test(test_ip_alloc)
wb_test(test_noise)
wb_test(test_state)
wb_test(test_http_key)
wb_test(test_magicdns)
